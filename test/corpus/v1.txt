enum Shape {
    Circle { radius: f64 },
    Rect { w: f64, h: f64 },
    Dot,
}

copy struct Point { x: i32; y: i32; }

fn parse(s: &str) -> Outcome<u8, E> {
    unchecked(index_checked, "i < len was checked") {
        if (s.len() == 0) { return Bad(E::Fail); }
    }
    return Good(1u8);
}

fn forward(s: &str) -> Outcome<i32, E> {
    let n = parse(s) else return;
    let mut m: u8 = 0;
    m = parse(s) else (e) { return Bad(e); };
    parse(s) else trap("bad");
    return Good(n as i32 + m as i32);
}

fn area(s: Shape) -> i32 {
    switch (s) {
        case Circle { radius: r } if (r > 0.0): { return 1; }
        case Rect { w, h: 4.0 }: { return 2; }
        case Dot: { return 0; }
    }
    return -1;
}

fn classify(p: Point) -> i32 {
    switch (p) {
        case Point { x: 0, y: y }: { return y; }
        default: { return -1; }
    }
    return -2;
}

macro all_true(conds: code...) -> expr {
    for (c : conds) {
        if (!c) { return false; }
    }
    return true;
}

    let n = #square(3);
    #all_true(true, n > 0);
