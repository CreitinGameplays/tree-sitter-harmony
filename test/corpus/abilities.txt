abilities Drawable {
    fn area(&self) -> i32;
    fn label(&self) -> &str;
}

struct Square { side: i32; }

methods Square {
    export fn area(&self) -> i32 { return self.side * self.side; }
    export fn label(&self) -> &str { return "square"; }
}

fn show<T: Drawable + ord>(item: &T) -> i32 {
    return item.area();
}

fn main() -> i32 {
    let sq = Square { .side = 5 };
    return show(&sq);
}
